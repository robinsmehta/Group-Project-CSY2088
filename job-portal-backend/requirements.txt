# requirements.txt
# Lists all Python packages required for the Job Portal backend.
# Install with: pip install -r requirements.txt

flask                # Core web framework
flask-sqlalchemy     # ORM for working with database models
flask-cors           # Cross-Origin Resource Sharing handling
pymysql              # MySQL database driver
python-dotenv        # Loads environment variables from .env files
flask-bcrypt         # Password hashing utility
flask-migrate        # Database migrations tool
pytest               # Automated testing framework
